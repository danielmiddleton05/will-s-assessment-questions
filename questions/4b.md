###### Option 4
The events are dependent — Reverses the definition. Students may think any real-world connection between runner assignment and failure implies dependence, even though the diagram shows no change in failure probability across runners.

def setup_style():
    """Configure matplotlib style to match scientific plotting standards."""
    plt.rcParams['font.family'] = 'DejaVu Sans'
    plt.rcParams['font.size'] = 10
    plt.rcParams['axes.titlesize'] = 14
    plt.rcParams['axes.titleweight'] = 'bold'
    plt.rcParams['axes.labelsize'] = 12
    plt.rcParams['axes.labelcolor'] = '#333333'
    plt.rcParams['xtick.labelsize'] = 10
    plt.rcParams['ytick.labelsize'] = 10
    plt.rcParams['xtick.color'] = '#666666'
    plt.rcParams['ytick.color'] = '#666666'
    plt.rcParams['axes.edgecolor'] = '#333333'
    plt.rcParams['axes.linewidth'] = 1.0
    plt.rcParams['grid.color'] = '#CCCCCC'
    plt.rcParams['grid.linewidth'] = 0.5
    plt.rcParams['figure.facecolor'] = 'white'
    plt.rcParams['axes.facecolor'] = 'white'

def draw_node_box(ax, x, y, text, box_color='#FFFFFF', edge_color='#333333'):
    """Draw a rectangular node with text."""
    ax.text(x, y, text, 
            ha='center', va='center', 
            fontsize=12, color='black',
            bbox=dict(boxstyle='square,pad=0.5', fc=box_color, ec=edge_color, lw=1.0))

def draw_branch(ax, start, end, label=None):
    """Draw a line between two points and add a label."""
    # Draw line
    ax.plot([start[0], end[0]], [start[1], end[1]], color='#333333', linewidth=1.5, zorder=1)
    
    # Add label
    if label:
        mid_x = (start[0] + end[0]) / 2
        mid_y = (start[1] + end[1]) / 2
        # Offset label slightly up or based on slope to avoid overlapping
        offset_y = 0.02 if end[1] > start[1] else 0.02
        if end[1] < start[1]: 
             offset_y = -0.04
        
        ax.text(mid_x, mid_y + offset_y, label, 
                ha='center', va='center', 
                fontsize=10, color='#333333',
                backgroundcolor='white') # clear background behind text

def plot_image():
    """Generate the scientific visualization."""
    setup_style()

    fig, ax = plt.subplots(figsize=(10.24, 10.24))

    # Turn off axes for the diagram look
    ax.axis('off')

    ax.set_xlim(0, 10)

    x_L1 = 4.5
    x_L2 = 8.0

    # Branches from Runner X

    y_L2_X_nofail = 6.0
    
    # Branches from Runner Y

    y_L2_Y_nofail = 2.0

    # --- Draw Edges (Branches) ---
    
    # Root to Runner X
    draw_branch(ax, (x_root, y_root), (x_L1, y_L1_top), "0.60")
    
    # Root to Runner Y
    draw_branch(ax, (x_root, y_root), (x_L1, y_L1_bot), "0.40")
    
    # Runner X to Fail
    draw_branch(ax, (x_L1 + 0.5, y_L1_top), (x_L2, y_L2_X_fail), "0.12")
    
    # Runner X to Not fail
    draw_branch(ax, (x_L1 + 0.5, y_L1_top), (x_L2, y_L2_X_nofail), "0.88")
    
    # Runner Y to Fail
    draw_branch(ax, (x_L1 + 0.5, y_L1_bot), (x_L2, y_L2_Y_fail), "0.12")

    plt.tight_layout()
    return fig

if __name__ == "__main__":
