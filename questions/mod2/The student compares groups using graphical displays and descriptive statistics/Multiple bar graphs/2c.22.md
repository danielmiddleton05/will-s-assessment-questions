##### Explanation
From the graph, the day shift error rates by department are 4, 3, 5, and 2 errors per 1,000 doses, so the day shift range is \(5 - 2 = 3\).  
The night shift error rates are 6, 2, 7, and 3 errors per 1,000 doses, so the night shift range is \(7 - 2 = 5\).  
Because \(5 > 3\), the range is larger for the night shift than for the day shift.

##### Distractors

###### Option 1
Students may swap which range belongs to which shift. After correctly computing both ranges (3 and 5), they assign 5 to the day shift and 3 to the night shift, reversing the comparison.

###### Option 2
Students may correctly compute the day shift range as \(5 - 2 = 3\) but misidentify the night shift maximum as 6 (Emergency) instead of 7 (Surgical Ward), yielding a night shift range of \(6 - 2 = 4\). This leads to an underestimate of the difference between the two ranges.

###### Option 4
Students may correctly compute the day shift range as \(5 - 2 = 3\) but misidentify the night shift maximum as 6 (Emergency) instead of 7 (Surgical Ward) and the night shift minimum as 3 (Pediatrics) instead of 2 (ICU), yielding a night shift range of \(6 - 3 = 3\). This leads to the incorrect conclusion that both ranges equal 3.

import matplotlib.pyplot as plt
import numpy as np

def setup_style():
    """Configure matplotlib style to match scientific plotting standards."""
    plt.rcParams['font.family'] = 'DejaVu Sans'
    plt.rcParams['font.size'] = 10
    plt.rcParams['axes.titlesize'] = 14
    plt.rcParams['axes.titleweight'] = 'bold'
    plt.rcParams['axes.titlecolor'] = '#333333'
    plt.rcParams['axes.labelsize'] = 12
    plt.rcParams['axes.labelcolor'] = '#333333'
    plt.rcParams['xtick.labelsize'] = 10
    plt.rcParams['ytick.labelsize'] = 10
    plt.rcParams['xtick.color'] = '#666666'
    plt.rcParams['ytick.color'] = '#666666'
    plt.rcParams['axes.edgecolor'] = '#333333'
    plt.rcParams['axes.linewidth'] = 1.0
    plt.rcParams['grid.color'] = '#CCCCCC'
    plt.rcParams['grid.linewidth'] = 0.5
    plt.rcParams['figure.facecolor'] = 'white'
    plt.rcParams['axes.facecolor'] = 'white'

def plot_image():
    """Generate the scientific visualization."""
    setup_style()

    # Data definition
    departments = ["Emergency", "ICU", "Surgical Ward", "Pediatrics"]
    day_shift = [5, 8, 4, 3]
    night_shift = [7, 12, 6, 4]
    
    # Parameters for grouping
    x = np.arange(len(departments))
    width = 0.35  # width of the bars

    fig, ax = plt.subplots(figsize=(10.24, 10.24))

    # Plot bars
    # Day shift (Blue)
    rects1 = ax.bar(x - width/2, day_shift, width, label='Day Shift', 
                    color='#1f77b4', zorder=3)
    # Night shift (Orange)
    rects2 = ax.bar(x + width/2, night_shift, width, label='Night Shift', 
                    color='#ff7f0e', zorder=3)

    # Styling and Labels
    ax.set_title("Medication Administration Error Rates by Department and Shift", pad=20)
    ax.set_xlabel("Department")
    ax.set_ylabel("Error Rate (per 1,000 doses)")
    
    # X-axis setup
    ax.set_xticks(x)
    ax.set_xticklabels(departments)
    ax.tick_params(axis='x', length=5, direction='out', colors='#666666')
    
    # Y-axis setup
    ax.set_ylim(0, 14)
    ax.set_yticks(np.arange(0, 15, 2))
    ax.tick_params(axis='y', length=5, direction='out', colors='#666666')

    # Grid setup (Horizontal only, behind data)
    ax.set_axisbelow(True)
    ax.grid(axis='y', linestyle='-', alpha=1.0)

    # Spines
    ax.spines['top'].set_visible(False)
    ax.spines['right'].set_visible(False)
    
    # Legend
    ax.legend(loc='upper right', frameon=False, fontsize=10)

    plt.tight_layout()
    return fig

if __name__ == "__main__":
    fig = plot_image()
    plt.savefig('plot.png', dpi=150, bbox_inches='tight')
    print("Image saved as plot.png")
