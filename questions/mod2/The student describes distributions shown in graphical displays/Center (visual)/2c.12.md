import matplotlib.pyplot as plt
import numpy as np

def setup_style():
    """Configure matplotlib style to match scientific plotting standards."""
    plt.rcParams['font.family'] = 'DejaVu Sans'
    plt.rcParams['font.size'] = 10
    plt.rcParams['axes.titlesize'] = 14
    plt.rcParams['axes.titleweight'] = 'bold'
    plt.rcParams['axes.labelsize'] = 12
    plt.rcParams['axes.labelcolor'] = '#333333'
    plt.rcParams['xtick.labelsize'] = 10
    plt.rcParams['ytick.labelsize'] = 10
    plt.rcParams['xtick.color'] = '#666666'
    plt.rcParams['ytick.color'] = '#666666'
    plt.rcParams['axes.edgecolor'] = '#333333'
    plt.rcParams['axes.linewidth'] = 1.0
    plt.rcParams['grid.color'] = '#CCCCCC'
    plt.rcParams['grid.linewidth'] = 0.5
    plt.rcParams['figure.facecolor'] = 'white'
    plt.rcParams['axes.facecolor'] = 'white'

def plot_image():
    """Generate the scientific visualization."""
    setup_style()

    # Create figure with specific size (10.24x10.24 inches)
    # Using a grid of 1 row x 4 columns
    fig, axes = plt.subplots(1, 4, figsize=(10.24, 10.24), sharey=True)
    
    # Department configurations
    # Urgent Care: Center ~35, Range 20-50
    # Radiology: Center ~15, Range 5-25
    # Outpatient Lab: Center ~20, Range 10-30
    # Pharmacy: Center ~40, Range 25-55
    departments = [
        {"name": "Urgent Care", "mean": 35, "std": 6},
        {"name": "Radiology", "mean": 15, "std": 4},
        {"name": "Outpatient Lab", "mean": 20, "std": 4},
        {"name": "Pharmacy", "mean": 40, "std": 6}
    ]

    # Generate consistent random data
    np.random.seed(42)
    sample_size = 500
    bins = np.linspace(0, 60, 13)  # Bins of width 5 for clear reading (0, 5, 10...)

    for i, (ax, dept) in enumerate(zip(axes, departments)):
        # Generate data
        data = np.random.normal(dept["mean"], dept["std"], sample_size)
        
        # Clip data to ensure it stays somewhat within realistic bounds for the visual
        data = np.clip(data, 0, 60)

        # Plot histogram
        ax.hist(data, bins=bins, color='#1f77b4', edgecolor='#333333', linewidth=0.8, rwidth=0.9)

        # Style Spines
        ax.spines['top'].set_visible(False)
        ax.spines['right'].set_visible(False)
        ax.spines['left'].set_color('#333333')
        ax.spines['bottom'].set_color('#333333')

        # Grid
        ax.set_axisbelow(True)
        ax.grid(axis='y', linestyle='-', alpha=0.7)

        # Labels and Titles
        ax.set_title(dept["name"], pad=15)
        ax.set_xticks([0, 10, 20, 30, 40, 50, 60])
        ax.set_xlim(0, 60)
        
        # Y-Axis label only on the first plot
        if i == 0:
            ax.set_ylabel("Frequency")
        else:
            # Hide y-axis spine/ticks for inner plots if desired, or just keep labels off
            # Here we keep ticks for readability but remove label
            pass

    # Global X-axis label
    # Since subplots are side-by-side, we place one label centered at the bottom
    fig.text(0.5, 0.04, 'Check-in Wait Time (minutes)', ha='center', va='center', 
             fontsize=12, color='#333333')

    # Adjust layout to prevent overlap and make room for global label
    plt.tight_layout()
    plt.subplots_adjust(bottom=0.1) # Add extra space at bottom for the global x-label

    return fig

if __name__ == "__main__":
    fig = plot_image()
    plt.savefig('plot.png', dpi=150, bbox_inches='tight')
    print("Image saved as plot.png")
