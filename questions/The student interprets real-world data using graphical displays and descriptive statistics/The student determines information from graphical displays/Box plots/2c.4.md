##### Explanation
Q3 (the third quartile) in a box plot represents the 75th percentile, meaning three-quarters of the data values fall at or below this point. In this box plot, Q3 is 60 minutes, which is shown by the right edge of the box. Therefore, three-quarters of the patients waited at or below 60 minutes. Q1 (30 minutes) represents the 25th percentile where only one-quarter of patients fall below. The median (40 minutes) represents the 50th percentile where half of patients fall below. The maximum (80 minutes) represents the longest wait time, where all patients fall at or below, not three-quarters.

##### Distractors
###### Option 1
Confuses Q1 with Q3 — Students may select Q1 (30 minutes) if they misunderstand quartile positions or confuse the left edge of the box with the right edge. Q1 actually represents the 25th percentile, meaning only one-quarter of patients waited below this time, not three-quarters.

###### Option 2
Confuses the median with Q3 — Students may select the median (40 minutes) if they misidentify the line inside the box as the point where three-quarters of the data falls. The median represents the 50th percentile, meaning half of patients waited below this time, not three-quarters.

###### Option 4
Confuses maximum with Q3 — Students may select the maximum (80 minutes) if they interpret "three-quarters waited at or below" as referring to the endpoint of the data. The maximum represents the longest wait time in the dataset, where all patients fall at or below, not three-quarters.

import matplotlib.pyplot as plt
import numpy as np

def setup_style():
    """Configure matplotlib style to match scientific plotting standards."""
    plt.rcParams['font.family'] = 'DejaVu Sans'
    plt.rcParams['font.size'] = 10
    plt.rcParams['axes.titlesize'] = 14
    plt.rcParams['axes.titleweight'] = 'bold'
    plt.rcParams['axes.labelsize'] = 12
    plt.rcParams['axes.labelcolor'] = '#333333'
    plt.rcParams['xtick.labelsize'] = 10
    plt.rcParams['ytick.labelsize'] = 10
    plt.rcParams['xtick.color'] = '#666666'
    plt.rcParams['ytick.color'] = '#666666'
    plt.rcParams['axes.edgecolor'] = '#333333'
    plt.rcParams['axes.linewidth'] = 1.0
    plt.rcParams['grid.color'] = '#CCCCCC'
    plt.rcParams['grid.linewidth'] = 0.5
    plt.rcParams['figure.facecolor'] = 'white'
    plt.rcParams['axes.facecolor'] = 'white'

def plot_image():
    """Generate the scientific visualization."""
    setup_style()

    fig, ax = plt.subplots(figsize=(10.24, 10.24))

    # Hide top and right spines
    ax.spines['top'].set_visible(False)
    ax.spines['right'].set_visible(False)

    # Data configuration for box plot (stats based on description)
    # minimum = 10, Q1 = 30, median = 40, Q3 = 60, maximum = 80
    stats = [{
        'whislo': 10,
        'q1': 30,
        'med': 40,
        'q3': 60,
        'whishi': 80,
        'label': ''  # No label needed for single box
    }]

    # Create horizontal box plot
    # patch_artist=True allows us to fill the box with color
    # vert=False makes it horizontal
    # widths controls the thickness of the box
    bp = ax.bxp(stats, vert=False, patch_artist=True, showfliers=False, 
                positions=[1], widths=0.4)

    # Styling the box plot elements
    # Box Color: Standard matplotlib blue (#1f77b4)
    for patch in bp['boxes']:
        patch.set_facecolor('#1f77b4')
        patch.set_edgecolor('black')
        patch.set_linewidth(1.0)

    # Line Color: Whiskers and caps solid black
    for element in ['whiskers', 'caps']:
        for line in bp[element]:
            line.set_color('black')
            line.set_linewidth(1.0)

    # Median line styling (solid black)
    for line in bp['medians']:
        line.set_color('black')
        line.set_linewidth(1.5)

    # Title and Labels
    ax.set_title("Emergency Room Wait Times", pad=20)
    ax.set_xlabel("Wait Time (minutes)", labelpad=10)

    # Axis configuration
    # X-Axis: 0 to 90, ticks every 10
    ax.set_xlim(0, 90)
    ax.set_xticks(np.arange(0, 91, 10))

    # Y-Axis: Hidden/Unlabeled, centering the plot
    ax.set_yticks([])
    ax.set_ylim(0.5, 1.5)
    
    # Hide the left spine as well since Y-axis is hidden/unlabeled for this chart type
    ax.spines['left'].set_visible(False)

    # Grid configuration
    # Vertical grid lines aligned with x-axis ticks, behind data
    ax.grid(axis='x', linestyle='-', alpha=1.0)
    ax.set_axisbelow(True)

    plt.tight_layout()
    return fig

if __name__ == "__main__":
    fig = plot_image()
    plt.savefig('plot.png', dpi=150, bbox_inches='tight')
    print("Image saved as plot.png")
