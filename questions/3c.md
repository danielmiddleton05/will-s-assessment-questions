###### Option 4
This option mistakes the response variable for a summary statistic. Students who select this may conflate the act of recording individual measurements with the process of computing an average, or may not recognize that time to return results is a variable observed across testers rather than a single calculated value.

import numpy as np

def setup_style():
    """Configure matplotlib style to match scientific plotting standards."""

    setup_style()

    fig, ax = plt.subplots(figsize=(10.24, 10.24))

    np.random.seed(42)

    # Cluster 1: Low wait, High satisfaction
