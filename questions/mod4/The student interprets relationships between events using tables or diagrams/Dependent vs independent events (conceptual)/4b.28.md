#### Task Description

A site reliability team models two events for automated software builds:

- Event A: The build job is assigned to Runner X.  
- Event B: The build fails.

The team summarizes recent builds with this probability tree:

![Probability tree diagram showing Runner X and Runner Y branches with fail and not-fail outcomes and probabilities](https://k3-production-bucket.s3.us-east-1.amazonaws.com/uploads/E4YwkJrqRkg5NaLtN_plot_(16).png)

Based on this diagram, how are Event A and Event B related?

#### Quiz Options

###### Option 1 Incorrect
The events are complementary.


###### Option 2 Incorrect
The events are disjoint.


###### Option 3 Correct
The events are independent.


###### Option 4 Incorrect
The events are dependent.

##### Explanation
The diagram shows the probability of a build failing is 0.12 whether the job is assigned to Runner X or Runner Y. Because the chance of failure does not change based on which runner is assigned, the events are independent.

##### Distractors
###### Option 1
The events are complementary — Confuses independence with “opposites.” A build being assigned to Runner X and a build failing are not opposites, and they can occur together.

###### Option 2
The events are disjoint — Confuses dependence with mutual exclusivity. A job can be assigned to Runner X and also fail, so the events are not mutually exclusive.

###### Option 4
The events are dependent — Reverses the definition. Students may think any real-world connection between runner assignment and failure implies dependence, even though the diagram shows no change in failure probability across runners.

import matplotlib.pyplot as plt
import matplotlib.patches as patches

def setup_style():
    """Configure matplotlib style to match scientific plotting standards."""
    plt.rcParams['font.family'] = 'DejaVu Sans'
    plt.rcParams['font.size'] = 10
    plt.rcParams['axes.titlesize'] = 14
    plt.rcParams['axes.titleweight'] = 'bold'
    plt.rcParams['axes.labelsize'] = 12
    plt.rcParams['axes.labelcolor'] = '#333333'
    plt.rcParams['xtick.labelsize'] = 10
    plt.rcParams['ytick.labelsize'] = 10
    plt.rcParams['xtick.color'] = '#666666'
    plt.rcParams['ytick.color'] = '#666666'
    plt.rcParams['axes.edgecolor'] = '#333333'
    plt.rcParams['axes.linewidth'] = 1.0
    plt.rcParams['grid.color'] = '#CCCCCC'
    plt.rcParams['grid.linewidth'] = 0.5
    plt.rcParams['figure.facecolor'] = 'white'
    plt.rcParams['axes.facecolor'] = 'white'

def draw_node_box(ax, x, y, text, box_color='#FFFFFF', edge_color='#333333'):
    """Draw a rectangular node with text."""
    ax.text(x, y, text, 
            ha='center', va='center', 
            fontsize=12, color='black',
            bbox=dict(boxstyle='square,pad=0.5', fc=box_color, ec=edge_color, lw=1.0))

def draw_branch(ax, start, end, label=None):
    """Draw a line between two points and add a label."""
    # Draw line
    ax.plot([start[0], end[0]], [start[1], end[1]], color='#333333', linewidth=1.5, zorder=1)
    
    # Add label
    if label:
        mid_x = (start[0] + end[0]) / 2
        mid_y = (start[1] + end[1]) / 2
        # Offset label slightly up or based on slope to avoid overlapping
        offset_y = 0.02 if end[1] > start[1] else 0.02
        if end[1] < start[1]: 
             offset_y = -0.04
        
        ax.text(mid_x, mid_y + offset_y, label, 
                ha='center', va='center', 
                fontsize=10, color='#333333',
                backgroundcolor='white') # clear background behind text

def plot_image():
    """Generate the scientific visualization."""
    setup_style()

    fig, ax = plt.subplots(figsize=(10.24, 10.24))

    # Turn off axes for the diagram look
    ax.axis('off')
    
    # Set coordinate limits
    ax.set_xlim(0, 10)
    ax.set_ylim(0, 10)

    # Define coordinates
    x_root = 1.0
    x_L1 = 4.5
    x_L2 = 8.0

    y_root = 5.0
    
    # Level 1 Y coordinates (Runner X top, Runner Y bottom)
    y_L1_top = 7.0
    y_L1_bot = 3.0
    
    # Level 2 Y coordinates (Fail top, Not fail bottom relative to parent)
    # Branches from Runner X
    y_L2_X_fail = 8.0
    y_L2_X_nofail = 6.0
    
    # Branches from Runner Y
    y_L2_Y_fail = 4.0
    y_L2_Y_nofail = 2.0

    # --- Draw Edges (Branches) ---
    
    # Root to Runner X
    draw_branch(ax, (x_root, y_root), (x_L1, y_L1_top), "0.60")
    
    # Root to Runner Y
    draw_branch(ax, (x_root, y_root), (x_L1, y_L1_bot), "0.40")
    
    # Runner X to Fail
    draw_branch(ax, (x_L1 + 0.5, y_L1_top), (x_L2, y_L2_X_fail), "0.12")
    
    # Runner X to Not fail
    draw_branch(ax, (x_L1 + 0.5, y_L1_top), (x_L2, y_L2_X_nofail), "0.88")
    
    # Runner Y to Fail
    draw_branch(ax, (x_L1 + 0.5, y_L1_bot), (x_L2, y_L2_Y_fail), "0.12")
    
    # Runner Y to Not fail
    draw_branch(ax, (x_L1 + 0.5, y_L1_bot), (x_L2, y_L2_Y_nofail), "0.88")

    # --- Draw Nodes ---
    
    # Root Node (Just a small point to indicate start)
    ax.scatter([x_root], [y_root], color='#333333', s=50, zorder=2)
    
    # Level 1 Nodes
    draw_node_box(ax, x_L1, y_L1_top, "Runner X")
    draw_node_box(ax, x_L1, y_L1_bot, "Runner Y")
    
    # Level 2 Nodes (Terminal)
    draw_node_box(ax, x_L2, y_L2_X_fail, "Fail")
    draw_node_box(ax, x_L2, y_L2_X_nofail, "Not fail")
    draw_node_box(ax, x_L2, y_L2_Y_fail, "Fail")
    draw_node_box(ax, x_L2, y_L2_Y_nofail, "Not fail")

    plt.tight_layout()
    return fig

if __name__ == "__main__":
    fig = plot_image()
    plt.savefig('plot.png', dpi=150, bbox_inches='tight')
    print("Image saved as plot.png")
