#### Task Description

**Given:** A community college is comparing two relationships from the same course. For each relationship, the institutional research office produced a scatter plot with a fitted line and recorded the **coefficient of determination** \(R^2\).

- **Analysis 1 (Office-hour visits → Final course percentage)**  
  $R^2 = 0.07$

- **Analysis 2 (Unexcused absences → Final course percentage)**  
  $R^2 = 0.69$

##### Explanation
The primary concept assessed is comparing two bivariate relationships in terms of direction, form, and strength using scatter plots and \(R^2\). Both scatter plots show an approximately linear pattern (no curvature), establishing the form. The direction is determined from the fitted lines in the scatter plots: Analysis 1 slopes upward (positive), while Analysis 2 slopes downward (negative). Strength is indicated by \(R^2\): a value of \(0.07\) means only 7% of the variation is explained, indicating a weak relationship for Analysis 1, while \(0.69\) means 69% is explained, indicating a strong relationship for Analysis 2. This is also visible in how tightly the points cluster around each line.

##### Distractors

###### Option 1
Swaps the direction of both relationships by misreading the slope of each fitted line in the scatter plots.

###### Option 2
Reverses the relative strengths by treating the low \(R^2\) in Analysis 1 as strong and the high \(R^2\) in Analysis 2 as weak.

###### Option 4
Correctly identifies direction and linear form but incorrectly labels Analysis 2 as weak, ignoring that \(R^2 = 0.69\) indicates a much stronger association than \(R^2 = 0.07\).

import matplotlib.pyplot as plt
import numpy as np

def setup_style():
    """Configure matplotlib style to match scientific plotting standards."""
    plt.rcParams['font.family'] = 'DejaVu Sans'
    plt.rcParams['font.size'] = 10
    plt.rcParams['axes.titlesize'] = 14
    plt.rcParams['axes.titleweight'] = 'bold'
    plt.rcParams['axes.labelsize'] = 12
    plt.rcParams['axes.labelcolor'] = '#333333'
    plt.rcParams['xtick.labelsize'] = 10
    plt.rcParams['ytick.labelsize'] = 10
    plt.rcParams['xtick.color'] = '#333333'
    plt.rcParams['ytick.color'] = '#333333'
    plt.rcParams['axes.edgecolor'] = '#333333'
    plt.rcParams['axes.linewidth'] = 1.0
    plt.rcParams['grid.color'] = '#CCCCCC'
    plt.rcParams['grid.linewidth'] = 0.5
    plt.rcParams['figure.facecolor'] = 'white'
    plt.rcParams['axes.facecolor'] = 'white'
    plt.rcParams['lines.linewidth'] = 2

def generate_data(n_points, x_range, intercept, slope, target_r2):
    """
    Generate synthetic data for regression with a specific R-squared target.
    """
    np.random.seed(42)  # For reproducibility
    
    # Generate X values
    x = np.random.uniform(x_range[0], x_range[1], n_points)
    
    # Calculate predicted Y (signal)
    y_pred = intercept + slope * x
    
    # Calculate variance of signal
    var_signal = np.var(y_pred)
    
    if target_r2 == 0 or var_signal == 0:
        noise = np.random.normal(0, 1, n_points)
    else:
        # Calculate required noise variance based on target R^2
        # R^2 = var_signal / (var_signal + var_noise)
        # var_noise = var_signal * (1/R^2 - 1)
        var_noise = var_signal * (1/target_r2 - 1)
        std_noise = np.sqrt(var_noise)
        noise = np.random.normal(0, std_noise, n_points)
    
    y = y_pred + noise
    
    return x, y

def plot_image():
    """Generate the scientific visualization."""
    setup_style()

    fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(10.24, 6))

    # --- Plot 1: Analysis 1 ---
    # Parameters: Intercept=74.2, Slope=0.5, R^2=0.07
    x1, y1 = generate_data(n_points=50, x_range=(0, 25), intercept=74.2, slope=0.5, target_r2=0.07)
    
    # Scatter plot
    ax1.scatter(x1, y1, color='#1f77b4', edgecolor='#333333', linewidth=0.8, alpha=0.9, zorder=3)
    
    # Regression Line
    x_line1 = np.array([0, 25])
    y_line1 = 74.2 + 0.5 * x_line1
    ax1.plot(x_line1, y_line1, color='#d62728', linestyle='-', zorder=4)
    
    # Styling
    ax1.set_title("Analysis 1")
    ax1.set_xlabel("Office-hour Visits")
    ax1.set_ylabel("Final Percentage")
    ax1.set_xlim(0, 25)
    ax1.set_ylim(50, 100)
    ax1.grid(True, zorder=0)
    
    # Spines
    ax1.spines['top'].set_visible(False)
    ax1.spines['right'].set_visible(False)
    ax1.set_axisbelow(True)

    # --- Plot 2: Analysis 2 ---
    # Parameters: Intercept=93.6, Slope=-1.8, R^2=0.69
    np.random.seed(123) 
    x2 = np.random.uniform(0, 20, 50)
    y_pred2 = 93.6 - 1.8 * x2
    var_signal2 = np.var(y_pred2)
    target_r2_2 = 0.69
    var_noise2 = var_signal2 * (1/target_r2_2 - 1)
    noise2 = np.random.normal(0, np.sqrt(var_noise2), 50)
    y2 = y_pred2 + noise2
    
    # Scatter plot
    ax2.scatter(x2, y2, color='#1f77b4', edgecolor='#333333', linewidth=0.8, alpha=0.9, zorder=3)
    
    # Regression Line
    x_line2 = np.array([0, 20])
    y_line2 = 93.6 - 1.8 * x_line2
    ax2.plot(x_line2, y_line2, color='#d62728', linestyle='-', zorder=4)
    
    # Styling
    ax2.set_title("Analysis 2")
    ax2.set_xlabel("Unexcused Absences")
    ax2.set_ylabel("Final Percentage")
    ax2.set_xlim(0, 20)
    ax2.set_ylim(50, 100)
    ax2.grid(True, zorder=0)
    
    # Spines
    ax2.spines['top'].set_visible(False)
    ax2.spines['right'].set_visible(False)
    ax2.set_axisbelow(True)

    plt.tight_layout(pad=3.0)
    return fig

if __name__ == "__main__":
    fig = plot_image()
    plt.savefig('plot.png', dpi=150, bbox_inches='tight')
    print("Image saved as plot.png")
