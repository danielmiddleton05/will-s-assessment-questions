##### Competency
The student analyzes relationships between variables in real-world contexts, including correlation and regression.

##### Evidence Statement

##### Aligned Content
No clear association (weak or absent pattern)

##### Explanation
The correct choice is **Scatterplot C** because it shows points scattered without any discernible upward trend, downward trend, or other consistent pattern. When a scatterplot looks like a random cloud of points, it indicates **no clear association** between the variables.

*(Image construction guidance: A shows a weak-to-moderate positive linear trend; B shows a moderate negative linear trend; C shows random scatter with no trend; D shows a clear non-linear pattern such as a U-shape or inverted U-shape.)*

##### Distractors

###### Option 1
**Scatterplot A** — Misinterprets a weak positive trend as “no relationship.”  
Students may focus on the spread of points and miss that the overall direction is upward.

###### Option 2
**Scatterplot B** — Confuses a negative association with no association.  
A downward pattern still indicates a relationship; it is just negative rather than positive.

###### Option 4
**Scatterplot D** — Treats a non-linear pattern as if it were “no association.”  
If the points form a curved shape (rather than a straight-line trend), there is still an association even though it is not linear.

import matplotlib.pyplot as plt
import numpy as np

def setup_style():
    """Configure matplotlib style to match scientific plotting standards."""
    plt.rcParams['font.family'] = 'DejaVu Sans'
    plt.rcParams['font.size'] = 10
    plt.rcParams['axes.titlesize'] = 14
    plt.rcParams['axes.titleweight'] = 'bold'
    plt.rcParams['axes.labelsize'] = 12
    plt.rcParams['axes.labelcolor'] = '#333333'
    plt.rcParams['xtick.labelsize'] = 10
    plt.rcParams['ytick.labelsize'] = 10
    plt.rcParams['xtick.color'] = '#666666'
    plt.rcParams['ytick.color'] = '#666666'
    plt.rcParams['xtick.direction'] = 'out'
    plt.rcParams['ytick.direction'] = 'out'
    plt.rcParams['axes.edgecolor'] = '#333333'
    plt.rcParams['axes.linewidth'] = 1.0
    plt.rcParams['grid.color'] = '#E0E0E0'
    plt.rcParams['grid.linewidth'] = 0.5
    plt.rcParams['grid.linestyle'] = '-'
    plt.rcParams['figure.facecolor'] = 'white'
    plt.rcParams['axes.facecolor'] = 'white'
    # Ensure grid is behind data
    plt.rcParams['axes.axisbelow'] = True

def generate_correlated_data(n_points, correlation_type):
    """Generate x, y data based on desired correlation type."""
    np.random.seed(42)  # For reproducibility, though called multiple times, we'll shift seed or use randomness
    
    # Base X values
    x = np.random.uniform(10, 90, n_points)
    noise = np.random.normal(0, 1, n_points)
    
    if correlation_type == 'mod_pos':
        # Moderate positive: slope ~ 1, moderate noise
        y = 0.8 * x + 20 + np.random.normal(0, 15, n_points)
    elif correlation_type == 'mod_neg':
        # Moderate negative: slope ~ -1, moderate noise
        y = -0.8 * x + 100 + np.random.normal(0, 15, n_points)
    elif correlation_type == 'none':
        # No association: random cloud
        x = np.random.uniform(10, 90, n_points)
        y = np.random.uniform(10, 90, n_points)
    elif correlation_type == 'strong_pos':
        # Strong positive: slope ~ 1, low noise
        y = 1.0 * x + 10 + np.random.normal(0, 5, n_points)
    else:
        y = x
        
    return x, y

def plot_image():
    """Generate the scientific visualization."""
    setup_style()

    # Create figure with 2x2 grid
    fig, axes = plt.subplots(2, 2, figsize=(10.24, 10.24))
    
    # Flatten axes array for easy iteration
    axs = axes.flatten()
    
    # Define plot configurations
    plot_configs = [
        {'label': 'A', 'type': 'mod_pos', 'seed': 101},
        {'label': 'B', 'type': 'mod_neg', 'seed': 102},
        {'label': 'C', 'type': 'none', 'seed': 103},
        {'label': 'D', 'type': 'strong_pos', 'seed': 104}
    ]
    
    # Common visuals
    scatter_color = '#1f77b4'
    n_points = 60
    
    for ax, config in zip(axs, plot_configs):
        # Set local seed for specific distribution
        np.random.seed(config['seed'])
        
        # Generate data
        x, y = generate_correlated_data(n_points, config['type'])
        
        # Plot scatter
        ax.scatter(x, y, 
                   c=scatter_color, 
                   s=100, 
                   alpha=1.0, 
                   edgecolors='#333333', 
                   linewidths=0.8,
                   zorder=3)
        
        # Styling
        ax.grid(True)
        
        # Hide top and right spines
        ax.spines['top'].set_visible(False)
        ax.spines['right'].set_visible(False)
        
        # Add label (A, B, C, D) in top-left
        # Using axes coordinates for consistent placement
        ax.text(-0.05, 1.05, config['label'], 
                transform=ax.transAxes, 
                fontsize=16, 
                fontweight='bold', 
                color='#333333',
                va='bottom', ha='right')

        # Remove individual axis labels to unclutter, 
        # or keep ticks but remove labels if we want a cleaner "small multiples" look.
        # However, description implies standard plots. 
        # Let's keep tick labels but ensure range is nice.
        ax.set_xlim(0, 100)
        ax.set_ylim(0, 120)

    # Common Labels
    # We use a large frame or fig.text to place common X and Y labels
    # Position: centered bottom for X, centered left (rotated) for Y
    fig.text(0.5, 0.04, 'Operational Metric', ha='center', va='center', fontsize=14, color='#333333')
    fig.text(0.04, 0.5, 'Weekly Profit ($)', ha='center', va='center', rotation='vertical', fontsize=14, color='#333333')

    # Adjust layout to make room for common labels
    plt.tight_layout(rect=[0.05, 0.05, 0.98, 0.95])
    
    return fig

if __name__ == "__main__":
    fig = plot_image()
    plt.savefig('plot.png', dpi=150, bbox_inches='tight')
    print("Image saved as plot.png")
